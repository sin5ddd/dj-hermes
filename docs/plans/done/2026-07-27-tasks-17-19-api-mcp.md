`src/api.rs`:
```rust
use crate::engine::Command;
use crate::song::parse_song;
use axum::{extract::State, http::StatusCode, response::sse::{Event, Sse},
           routing::{get, post, put}, Json, Router};
use crossbeam::channel::Sender;
use serde::{Deserialize, Serialize};
use std::convert::Infallible;
use std::path::PathBuf;
use std::sync::{Arc, Mutex};

#[derive(Clone)]
pub struct AppState {
    pub tx: Sender<Command>,
    pub status: Arc<Mutex<StatusInfo>>,
}

#[derive(Clone, Default, Serialize)]
pub struct StatusInfo {
    pub deck_a: Option<String>,   // 曲タイトル
    pub deck_b: Option<String>,
    pub bpm: f64,
    pub playing: bool,
}

#[derive(Deserialize)]
pub struct CodeReq { pub code: String, pub deck: Option<String> }
#[derive(Deserialize)]
pub struct LoadReq { pub path: String, pub deck: String }
#[derive(Deserialize)]
pub struct XFadeReq { pub to: String, pub bars: Option<u32> }
#[derive(Deserialize)]
pub struct BpmReq { pub bpm: f64 }
#[derive(Deserialize)]
pub struct MuteReq { pub deck: String, pub track: String, pub muted: bool }
#[derive(Serialize)]
pub struct ErrRes { pub error: String }

fn bad(e: String) -> (StatusCode, Json<ErrRes>) {
    (StatusCode::BAD_REQUEST, Json(ErrRes { error: e }))
}

/// LLM 向けワンショット: 単一パターンコードを曲として即ロード
async fn put_code(State(s): State<AppState>, Json(r): Json<CodeReq>)
    -> Result<StatusCode, (StatusCode, Json<ErrRes>)> {
    let deck = deck_idx(r.deck.as_deref().unwrap_or("A"))?;
    let song = crate::song::Song {
        title: "api".into(), bpm: None, path: String::new(),
        tracks: vec![crate::song::Track {
            name: "main".into(), muted: false,
            code: crate::code::parse_code(&r.code).map_err(bad)?,
        }],
    };
    s.tx.send(Command::LoadSong { deck, song }).map_err(|e| bad(e.to_string()))?;
    Ok(StatusCode::ACCEPTED)   // 202: 次の小節から適用
}

async fn load_song(State(s): State<AppState>, Json(r): Json<LoadReq>)
    -> Result<StatusCode, (StatusCode, Json<ErrRes>)> {
    let deck = deck_idx(&r.deck)?;
    let text = std::fs::read_to_string(PathBuf::from(&r.path)).map_err(|e| bad(e.to_string()))?;
    let song = parse_song(&text, &r.path).map_err(bad)?;
    s.tx.send(Command::LoadSong { deck, song }).map_err(|e| bad(e.to_string()))?;
    Ok(StatusCode::ACCEPTED)
}

async fn xfade(State(s): State<AppState>, Json(r): Json<XFadeReq>)
    -> Result<StatusCode, (StatusCode, Json<ErrRes>)> {
    s.tx.send(Command::XFade { to_deck: deck_idx(&r.to)?, bars: r.bars.unwrap_or(4) })
        .map_err(|e| bad(e.to_string()))?;
    Ok(StatusCode::ACCEPTED)
}

async fn set_bpm(State(s): State<AppState>, Json(r): Json<BpmReq>)
    -> Result<StatusCode, (StatusCode, Json<ErrRes>)> {
    s.tx.send(Command::SetBpm(r.bpm)).map_err(|e| bad(e.to_string()))?;
    Ok(StatusCode::ACCEPTED)
}

async fn mute(State(s): State<AppState>, Json(r): Json<MuteReq>)
    -> Result<StatusCode, (StatusCode, Json<ErrRes>)> {
    s.tx.send(Command::SetTrackMute {
        deck: deck_idx(&r.deck)?, track: r.track, muted: r.muted,
    }).map_err(|e| bad(e.to_string()))?;
    Ok(StatusCode::ACCEPTED)
}

async fn hush(State(s): State<AppState>) -> StatusCode {
    s.tx.send(Command::Hush).ok();
    s.status.lock().unwrap().playing = false;
    StatusCode::NO_CONTENT
}

async fn get_status(State(s): State<AppState>) -> Json<StatusInfo> {
    Json(s.status.lock().unwrap().clone())
}

async fn events(State(s): State<AppState>)
    -> Sse<impl tokio_stream::Stream<Item = Result<Event, Infallible>>> {
    let st = async_stream::stream! {
        let mut last = String::new();
        loop {
            let cur = serde_json::to_string(&*s.status.lock().unwrap()).unwrap();
            if cur != last { last = cur.clone(); yield Ok(Event::default().data(cur)); }
            tokio::time::sleep(std::time::Duration::from_millis(250)).await;
        }
    };
    Sse::new(st)
}

fn deck_idx(s: &str) -> Result<usize, (StatusCode, Json<ErrRes>)> {
    match s { "A" | "a" => Ok(0), "B" | "b" => Ok(1), _ => Err(bad(format!("deck must be A or B: {s}"))) }
}

pub fn router(state: AppState) -> Router {
    Router::new()
        .route("/health", get(|| async { "ok" }))
        .route("/code", put(put_code))
        .route("/song/load", post(load_song))
        .route("/xfade", post(xfade))
        .route("/bpm", post(set_bpm))
        .route("/mute", post(mute))
        .route("/hush", post(hush))
        .route("/status", get(get_status))
        .route("/events", get(events))
        .with_state(state)
}

pub async fn serve(state: AppState, port: u16) {
    let listener = tokio::net::TcpListener::bind(("127.0.0.1", port)).await.unwrap();
    println!("API on http://127.0.0.1:{port}");
    axum::serve(listener, router(state)).await.unwrap();
}
```

### Task 18: MCP server（rmcp, stdio）

**Objective:** Hermes から MCP ツールとして直接操作できるようにする。`strudel-rs mcp` サブコマンドで起動。

**Files:**
- Modify: `strudel-rs/Cargo.toml`（rmcp）
- Create: `strudel-rs/src/mcp.rs`
- Modify: `strudel-rs/src/main.rs`（サブコマンド分岐）

**Step 1: 設計**

MCP サーバは**別プロセス**として stdio で動き、内部で HTTP API (127.0.0.1:17878) に転送する薄いブリッジにする（audio プロセスと MCP プロセスを分離し、MCP 側のクラッシュが演奏に影響しない設計）。

提供ツール:
- `strudel_set_code(code, deck?)` — 単一パターンをロード
- `strudel_load_song(path, deck)` — 曲ファイルロード
- `strudel_xfade(to, bars?)` — デッキ切替
- `strudel_set_bpm(bpm)`
- `strudel_mute(deck, track, muted)`
- `strudel_hush()`
- `strudel_status()` — 現在状態の取得

`Cargo.toml` 追加: `rmcp = { version = "0.3", features = ["server", "transport-io"] }`, `reqwest = { version = "0.12", features = ["json", "blocking"] }`（ブリッジ用）

実装は rmcp の `#[tool]` マクロで各ツールを定義し、本体は reqwest::blocking で `http://127.0.0.1:17878` の対応エンドポイントへ POST/PUT するだけの形にする（各 ~15 行）。

**Step 2: 検証**

{"jsonrpc":"2.0","id":2,"method":"tools/list"}

### Task 19: エラーハンドリング総仕上げ

**Objective:** どの入力経路でも、壊れた入力は「現行継続 + 明確なエラー報告」になることを保証。

**Files:**
- Modify: `strudel-rs/src/api.rs`, `strudel-rs/src/repl.rs`（テスト・挙動確認）

**Step 1: チェックリスト実装**

- 曲ファイルのパースエラー → watcher は現行曲を継続（Task 15 で実装済み）→ API の `/song/load` も 400 を返し pending に積まないことをテスト
- 単発コードのパースエラー → `/code` は 400、現行曲継続
- 存在しないサンプル名 → そのイベントだけ捨てる（Deck::process で実装済み）→ 警告を stderr に出すかどうか: audio スレッドでは出力不可なので、スケジュール時に未知音名を集計し SSE 経由で通知（簡易: 無視でも可。実装時に判断し追加メモへ）
- REPL の `:load` で存在しないファイル → エラー表示のみ
